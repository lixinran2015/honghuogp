# 2连板票查找方案

## 一、需求分析

根据股吧人气榜中的股票，查找2连板票（之前2天都是涨停）。

### 1.1 业务定义
- **2连板**：连续2个交易日都涨停
- **涨停判断标准**：
  - 主板/中小板：涨幅 ≥ 9.5%（考虑四舍五入）
  - 创业板/科创板：涨幅 ≥ 19.5%（20%涨停板）
  - 或者：`close / pre_close >= 1.099`（主板）或 `>= 1.199`（创业板）

### 1.2 数据来源
- **股吧人气榜**：`FactGubaPopularityRank` 表
- **历史价格数据**：`FactDailyPriceQfq` 表（前复权价格）
- **股票基本信息**：`DimStock` 表（用于判断主板/创业板）

## 二、技术方案

### 2.1 数据查询流程

```
1. 获取最新股吧人气榜股票列表
   ↓
2. 查询这些股票最近3个交易日的数据（需要3天来判断连续2天）
   ↓
3. 判断每只股票是否连续2天涨停
   ↓
4. 返回2连板股票列表
```

### 2.2 涨停判断逻辑

```python
def is_limit_up(ts_code: str, trade_date: date, close: float, pre_close: float) -> bool:
    """
    判断是否涨停
    
    Args:
        ts_code: 股票代码（如 000001.SZ）
        trade_date: 交易日期
        close: 收盘价
        pre_close: 前一日收盘价
    
    Returns:
        bool: 是否涨停
    """
    if pre_close <= 0:
        return False
    
    # 判断主板/创业板
    is_cyb = ts_code.startswith('300') or ts_code.startswith('688')
    
    if is_cyb:
        # 创业板/科创板：20%涨停
        limit_up_ratio = 1.199  # 19.9%（考虑四舍五入）
    else:
        # 主板/中小板：10%涨停
        limit_up_ratio = 1.099  # 9.9%（考虑四舍五入）
    
    # 判断是否涨停：收盘价 / 前收盘价 >= 涨停比例
    return (close / pre_close) >= limit_up_ratio
```

### 2.3 2连板判断逻辑

```python
def is_2_consecutive_limit_up(ts_code: str, trade_date: date) -> bool:
    """
    判断是否2连板（连续2天涨停）
    
    Args:
        ts_code: 股票代码
        trade_date: 查询日期（今天）
    
    Returns:
        bool: 是否2连板
    """
    # 获取最近3个交易日的数据（今天、昨天、前天）
    # 判断：昨天和前天是否都涨停
    # 注意：如果今天是交易日，需要排除今天的数据
    
    # 查询最近3个交易日
    recent_days = get_recent_trading_days(trade_date, count=3)
    
    if len(recent_days) < 2:
        return False
    
    # 获取昨天和前天的数据
    yesterday = recent_days[-2]  # 倒数第2个（昨天）
    day_before = recent_days[-3] if len(recent_days) >= 3 else None  # 倒数第3个（前天）
    
    if not day_before:
        return False
    
    # 查询价格数据
    yesterday_data = get_price_data(ts_code, yesterday)
    day_before_data = get_price_data(ts_code, day_before)
    
    if not yesterday_data or not day_before_data:
        return False
    
    # 判断昨天是否涨停（需要前天的收盘价作为前收盘价）
    is_yesterday_limit_up = is_limit_up(
        ts_code, 
        yesterday, 
        yesterday_data['close'], 
        day_before_data['close']  # 前天的收盘价是昨天的前收盘价
    )
    
    # 判断前天是否涨停（需要大前天的收盘价）
    day_before_yesterday = recent_days[-4] if len(recent_days) >= 4 else None
    if not day_before_yesterday:
        return False
    
    day_before_yesterday_data = get_price_data(ts_code, day_before_yesterday)
    if not day_before_yesterday_data:
        return False
    
    is_day_before_limit_up = is_limit_up(
        ts_code,
        day_before,
        day_before_data['close'],
        day_before_yesterday_data['close']
    )
    
    # 2连板：昨天和前天都涨停
    return is_yesterday_limit_up and is_day_before_limit_up
```

## 三、实现方案

### 3.1 创建API接口

**文件路径**：`backend/api/startup/limit_up_2days.py`

```python
"""
2连板票查找API
根据股吧人气榜查找2连板股票
"""
from fastapi import APIRouter, HTTPException, Query
from typing import Optional, List, Dict
from datetime import datetime, date, timedelta
import logging
from sqlalchemy import and_, func, text

from data_warehouse.service.warehouse_service import WarehouseService
from data_warehouse.models.guba_popularity import FactGubaPopularityRank
from data_warehouse.models.generated_models import FactDailyPriceQfq
from data_warehouse.models.orm_classes import DimStock
from backend.api.social.guba_popularity_cache import get_popularity_stocks_cached

router = APIRouter()
logger = logging.getLogger(__name__)


@router.get("/limit-up-2days")
async def find_2_consecutive_limit_up(
    trade_date: Optional[str] = Query(None, description="查询日期，格式YYYY-MM-DD，默认今天"),
    min_rank: Optional[int] = Query(None, description="最低排名（筛选人气榜范围）"),
    max_rank: Optional[int] = Query(100, description="最高排名（默认前100名）")
) -> Dict:
    """
    查找股吧人气榜中的2连板股票
    
    Args:
        trade_date: 查询日期
        min_rank: 最低排名
        max_rank: 最高排名
    
    Returns:
        Dict: {
            'success': bool,
            'data': List[Dict],  # 2连板股票列表
            'count': int,
            'query_date': str
        }
    """
    try:
        ws = WarehouseService()
        session = ws.get_session()
        
        try:
            # 1. 确定查询日期
            if trade_date:
                query_date = datetime.strptime(trade_date, "%Y-%m-%d").date()
            else:
                query_date = datetime.now().date()
            
            # 2. 获取股吧人气榜股票列表
            popularity_stocks = _get_popularity_stocks(session, query_date, min_rank, max_rank)
            
            if not popularity_stocks:
                return {
                    'success': True,
                    'data': [],
                    'count': 0,
                    'query_date': query_date.isoformat(),
                    'message': '未找到人气榜数据'
                }
            
            logger.info(f"📊 从人气榜获取 {len(popularity_stocks)} 只股票，开始查找2连板...")
            
            # 3. 查找2连板股票
            limit_up_2days_stocks = _find_2_consecutive_limit_up(session, popularity_stocks, query_date)
            
            return {
                'success': True,
                'data': limit_up_2days_stocks,
                'count': len(limit_up_2days_stocks),
                'query_date': query_date.isoformat(),
                'popularity_count': len(popularity_stocks)
            }
            
        finally:
            session.close()
            
    except Exception as e:
        logger.error(f"查找2连板股票失败: {e}", exc_info=True)
        raise HTTPException(status_code=500, detail=f"查找失败: {str(e)}")


def _get_popularity_stocks(session, query_date: date, min_rank: Optional[int], max_rank: int) -> List[str]:
    """获取股吧人气榜股票列表"""
    query = session.query(FactGubaPopularityRank.ts_code).filter(
        FactGubaPopularityRank.crawl_date == query_date
    )
    
    if min_rank is not None:
        query = query.filter(FactGubaPopularityRank.rank_position >= min_rank)
    if max_rank is not None:
        query = query.filter(FactGubaPopularityRank.rank_position <= max_rank)
    
    results = query.all()
    return [row[0] for row in results]


def _find_2_consecutive_limit_up(session, ts_codes: List[str], query_date: date) -> List[Dict]:
    """
    查找2连板股票
    
    Args:
        session: 数据库会话
        ts_codes: 股票代码列表
        query_date: 查询日期
    
    Returns:
        List[Dict]: 2连板股票列表
    """
    results = []
    
    # 获取最近5个交易日（确保有足够数据）
    trading_dates = _get_recent_trading_dates(session, query_date, count=5)
    
    if len(trading_dates) < 3:
        logger.warning(f"交易日数据不足，无法判断2连板")
        return results
    
    # 昨天和前天
    yesterday = trading_dates[-2]  # 倒数第2个
    day_before = trading_dates[-3]  # 倒数第3个
    
    # 批量查询价格数据
    price_data = _batch_get_price_data(session, ts_codes, [day_before, yesterday])
    
    # 批量查询股票基本信息（用于判断主板/创业板）
    stock_info = _batch_get_stock_info(session, ts_codes)
    
    for ts_code in ts_codes:
        try:
            # 获取价格数据
            day_before_data = price_data.get(ts_code, {}).get(day_before)
            yesterday_data = price_data.get(ts_code, {}).get(yesterday)
            
            if not day_before_data or not yesterday_data:
                continue
            
            # 判断是否2连板
            if _is_2_consecutive_limit_up(
                session, 
                ts_code, 
                day_before, 
                yesterday, 
                day_before_data, 
                yesterday_data,
                stock_info.get(ts_code)
            ):
                # 获取人气榜排名信息
                rank_info = _get_rank_info(session, ts_code, query_date)
                
                results.append({
                    'ts_code': ts_code,
                    'name': stock_info.get(ts_code, {}).get('name', ''),
                    'rank_position': rank_info.get('rank_position'),
                    'rank_change': rank_info.get('rank_change'),
                    'yesterday_close': float(yesterday_data['close']),
                    'day_before_close': float(day_before_data['close']),
                    'yesterday_change_pct': float(yesterday_data.get('change_pct', 0)),
                    'day_before_change_pct': float(day_before_data.get('change_pct', 0)),
                    'yesterday_date': yesterday.isoformat(),
                    'day_before_date': day_before.isoformat()
                })
                
        except Exception as e:
            logger.warning(f"处理 {ts_code} 失败: {e}")
            continue
    
    # 按人气榜排名排序
    results.sort(key=lambda x: x['rank_position'] if x['rank_position'] else 999)
    
    return results


def _is_2_consecutive_limit_up(
    session,
    ts_code: str,
    day_before: date,
    yesterday: date,
    day_before_data: Dict,
    yesterday_data: Dict,
    stock_info: Optional[Dict]
) -> bool:
    """
    判断是否2连板
    
    Args:
        session: 数据库会话
        ts_code: 股票代码
        day_before: 前天的日期
        yesterday: 昨天的日期
        day_before_data: 前天的价格数据
        yesterday_data: 昨天的价格数据
        stock_info: 股票基本信息
    
    Returns:
        bool: 是否2连板
    """
    # 判断主板/创业板
    is_cyb = _is_cyb_stock(ts_code, stock_info)
    limit_up_ratio = 1.199 if is_cyb else 1.099  # 创业板20%，主板10%
    
    # 获取大前天的收盘价（用于判断前天是否涨停）
    trading_dates = _get_recent_trading_dates(session, day_before, count=3)
    if len(trading_dates) < 2:
        return False
    
    day_before_yesterday = trading_dates[-2]  # 大前天
    day_before_yesterday_data = _get_price_data(session, ts_code, day_before_yesterday)
    
    if not day_before_yesterday_data:
        return False
    
    # 判断前天是否涨停
    day_before_close = float(day_before_data['close'])
    day_before_yesterday_close = float(day_before_yesterday_data['close'])
    
    if day_before_yesterday_close <= 0:
        return False
    
    is_day_before_limit_up = (day_before_close / day_before_yesterday_close) >= limit_up_ratio
    
    # 判断昨天是否涨停
    yesterday_close = float(yesterday_data['close'])
    day_before_close_for_yesterday = day_before_close  # 前天的收盘价是昨天的前收盘价
    
    if day_before_close_for_yesterday <= 0:
        return False
    
    is_yesterday_limit_up = (yesterday_close / day_before_close_for_yesterday) >= limit_up_ratio
    
    # 2连板：前天和昨天都涨停
    return is_day_before_limit_up and is_yesterday_limit_up


def _is_cyb_stock(ts_code: str, stock_info: Optional[Dict]) -> bool:
    """判断是否创业板/科创板股票"""
    # 方法1：根据代码前缀
    if ts_code.startswith('300.') or ts_code.startswith('688.'):
        return True
    
    # 方法2：根据股票信息
    if stock_info and stock_info.get('market') in ['创业板', '科创板']:
        return True
    
    return False


def _get_recent_trading_dates(session, end_date: date, count: int = 5) -> List[date]:
    """获取最近N个交易日"""
    query = session.query(
        func.distinct(FactDailyPriceQfq.trade_date)
    ).filter(
        FactDailyPriceQfq.trade_date <= end_date
    ).order_by(
        FactDailyPriceQfq.trade_date.desc()
    ).limit(count)
    
    results = query.all()
    dates = sorted([row[0] for row in results])
    return dates


def _batch_get_price_data(session, ts_codes: List[str], dates: List[date]) -> Dict[str, Dict[date, Dict]]:
    """批量获取价格数据"""
    query = session.query(FactDailyPriceQfq).filter(
        and_(
            FactDailyPriceQfq.ts_code.in_(ts_codes),
            FactDailyPriceQfq.trade_date.in_(dates)
        )
    )
    
    results = query.all()
    
    data = {}
    for row in results:
        if row.ts_code not in data:
            data[row.ts_code] = {}
        data[row.ts_code][row.trade_date] = {
            'close': row.close,
            'change_pct': row.change_pct,
            'amount': row.amount,
            'turnover_rate': row.turnover_rate
        }
    
    return data


def _get_price_data(session, ts_code: str, trade_date: date) -> Optional[Dict]:
    """获取单只股票的价格数据"""
    query = session.query(FactDailyPriceQfq).filter(
        and_(
            FactDailyPriceQfq.ts_code == ts_code,
            FactDailyPriceQfq.trade_date == trade_date
        )
    ).first()
    
    if not query:
        return None
    
    return {
        'close': query.close,
        'change_pct': query.change_pct,
        'amount': query.amount,
        'turnover_rate': query.turnover_rate
    }


def _batch_get_stock_info(session, ts_codes: List[str]) -> Dict[str, Dict]:
    """批量获取股票基本信息"""
    query = session.query(DimStock).filter(
        DimStock.ts_code.in_(ts_codes)
    )
    
    results = query.all()
    
    data = {}
    for row in results:
        data[row.ts_code] = {
            'name': row.name,
            'market': row.market if hasattr(row, 'market') else None
        }
    
    return data


def _get_rank_info(session, ts_code: str, query_date: date) -> Dict:
    """获取人气榜排名信息"""
    query = session.query(FactGubaPopularityRank).filter(
        and_(
            FactGubaPopularityRank.ts_code == ts_code,
            FactGubaPopularityRank.crawl_date == query_date
        )
    ).first()
    
    if not query:
        return {'rank_position': None, 'rank_change': None}
    
    return {
        'rank_position': query.rank_position,
        'rank_change': query.rank_change
    }
```

## 五、优化建议

### 5.1 性能优化
1. **批量查询**：使用 `IN` 查询批量获取价格数据，减少数据库查询次数
2. **缓存交易日列表**：缓存最近交易日列表，避免重复查询
3. **索引优化**：确保 `FactDailyPriceQfq` 表的 `(ts_code, trade_date)` 有索引

### 5.2 功能扩展
1. **N连板查找**：支持查找3连板、4连板等
2. **历史回溯**：支持查询历史某一天的2连板股票
3. **统计信息**：返回2连板股票的数量、占比等统计信息

### 5.3 数据验证
1. **停牌处理**：排除停牌股票
2. **ST股票**：可选是否包含ST股票
3. **新股处理**：新股可能没有足够的历史数据
